
cold_boot:
; ****************************************************************************
; ****************************************************************************
;
;                       Odyssey II BIOS Disassembly
;                       ===========================
;
;                    by Paul Robson (AutismUk@aol.com)
;
;       Version 1.0     (26-Nov-1997)
;
;
;       To do : 0293 - 02C2 - What is this all about ?
;               03B1 - 03E9 - Some more mysterious data processing
;
; ****************************************************************************
;
;       BIOS ROM Routine Addresses:
;
;               $000    Cold Boot
;               $003    External T0 Interrupt
;               $007    Timer/Clock Interrupt
;               $009    VBlank Interrupt Routine 1
;               $01A    VBlank Interrupt Routine 2 (Collision & Clock)
;               $044    VBlank Interrupt Routine 3 (Tune)
;               $089    RAM to VDC VBlank Copying Check Code
;               $0A3    Copying Code
;               $0B0    Keyboard Routine
;               $0E7    Set up VDC Access
;               $0EC    Set up RAM Access
;               $0F1    Reset
;               $11C    Display Off
;               $127    Display On
;               $132    Enable Data Copy next VSYNC
;               $13D    Get Keystroke
;               $14B    Character/Colour Translation
;               $16B    Clear all Characters
;               $176    Wait for Interrupt
;               $17C    Display 2 digit BCD Characters
;               $1A2    Start Tune
;               $1B0    Up/Down Counter
;               $23A    Set up Quad Score Characters
;               $261    Translate & Copy Character & Colour
;               $26A    Bit test
;               $280    Bit clear
;               $28A    Bit set
;               $293    [Unknown]
;               $2C3    Select Game
;               $300    Frequency Data
;               $34A    Tune Data
;               $376    Keyboard In routine (end)
;               $37E    Miscellaneous Interrupt Handlers for Banked Roms
;               $38F    Read Joystick
;               $3B1    [Unknown]
;               $3CF    [Unknown]
;               $3EA    Character Write
;
; ****************************************************************************
;
;       Vectors in Odyssey II ROM Cartridges.
;
;               $400    Restart
;               $402    VBlank (External) Interrupt
;               $404    Timer/Clock Interrupt
;               $406    VBlank Routine Vector (if $402 = JMP $009)
;               $408    End of Select Game, A = Selection
;               $40A    Continuation of VBlank (if $406 = JMP $01A)
;
; ****************************************************************************
0x000: 84 00  JMP restart                         ; start the game
0x002: 00     NOP

external_t0_interrupt:
0x003: 84 02  JMP vblank_external_interrupt       ; external interrupt ?
0x005: 00     NOP
0x006: 00     NOP

timer_clock_interrupt:
0x007: 84 04  JMP timer_clock_interrupt_cart      ; timer/clock interrupt

vblank_interrupt_routine_1:
; ****************************************************************************
;
;       External Interrupt are often redirected here, by 0402 = Jmp 0009.
;       This damages Registers 5 and 6 of the base set.
;       It completes by going to $0406 if VBLANK is on, returning otherwise.
;       (Presumably this is timeable VBlank Code.)
;
; ****************************************************************************
0x009: c5     SEL RB0                             ; back to main bank
0x00a: ad     MOV R5,A                            ; save A in R5
0x00b: 09     IN A,P1                             ; read enable/switch status
0x00c: ae     MOV R6,A                            ; save in r6
0x00d: 14 e7  CALL set_up_vdc_access              ; access the VDC
0x00f: b8 a1  MOV R0,##0xa1                       ; read $0A1 in VDC
0x011: 80     MOVX A,@R0
0x012: 72 18  JB3 loc_0018                        ; in VSYNC ?

loc_0014:
0x014: fe     MOV A,R6                            ; get the old switch status
0x015: 39     OUTL P1,A                           ; set the enable/switch status
0x016: fd     MOV A,R5                            ; restore A
0x017: 93     RETR                                ; return from interrupt

loc_0018:
0x018: 84 06  JMP vblank_routine_vector           ; VSYNC vector

vblank_interrupt_routine_2:
; ****************************************************************************
;
;       Alternative Vertical Sync Routine. Above routine can
;       be directed here by $406 = JMP $01A. Completes by going
;       to $408.
;
;       Registers : 3D = Collision,3E = Clock,3F = Status
;
;       3F bit 7 if set causes copy from RAM to VDC, see later.
;       3F bit 6 causes delay in passing on to $408 (for tune player)
;
; ****************************************************************************
0x01a: a5     CLR F1                              ; Set F1
0x01b: b5     CPL F1
0x01c: b9 a2  MOV R1,##0xa2                       ; R1 = $A2 (Collision)
0x01e: b8 3d  MOV R0,##0x3d                       ; R0 = $3D ?
0x020: 81     MOVX A,@R1                          ; A = Collision Register

loc_0021:
0x021: a0     MOV @R0,A                           ; put in R$3D
0x022: 18     INC R0                              ; increment R0 (now $3E)
0x023: 10     INC @R0                             ; increment R$3E (Clock)
0x024: f0     MOV A,@R0
0x025: 53 3f  ANL A,##0x3f                        ; mask $3E with 00111111
0x027: d3 3c  XRL A,##0x3c                        ; zero now if $3E was 60(dec)
0x029: 96 2f  JNZ loc_002f                        ; if non zero skip
0x02b: f0     MOV A,@R0                           ; wrap it round back to zero
0x02c: 53 c0  ANL A,##0xc0
0x02e: a0     MOV @R0,A

loc_002f:
0x02f: 18     INC R0                              ; look at $3F which is some
0x030: f0     MOV A,@R0                           ; sort of status flag
0x031: 37     CPL A
0x032: f2 3a  JB7 loc_003a                        ; if bit 7 of 3F was set
0x034: 37     CPL A                               ; clear bit 7
0x035: 53 7f  ANL A,##0x7f
0x037: a0     MOV @R0,A                           ; update register $3F
0x038: 14 89  CALL ram_to_vdc_vblank_copying_check ; do the data copy

loc_003a:
0x03a: b9 3f  MOV R1,##0x3f                       ; look at R$3F but 6
0x03c: f1     MOV A,@R1
0x03d: 37     CPL A
0x03e: d2 14  JB6 loc_0014                        ; if its clear, exit
0x040: eb 14  DJNZ R3,loc_0014                    ; don't do user routine yet.
0x042: 84 0a  JMP continuation_of_vblank          ; call user routine

vblank_interrupt_routine_3:
; ****************************************************************************
;
;       Yet another VSYNC Routine. Continuation by setting $40A to
;       jmp $044. Doesn't seem very popular..... I think it is some
;       sort of sound playing routine - it is unpacking a tune from
;       $0300
;
; ****************************************************************************
0x044: fc     MOV A,R4                            ; A= R4 (delay from above ?)
0x045: e3     MOVP3 A,@A                          ; Read Data Table into A
0x046: a9     MOV R1,A                            ; and R1 [300+R4]
0x047: 1c     INC R4
0x048: fc     MOV A,R4
0x049: e3     MOVP3 A,@A                          ; R2 = [301+R4], R4 inc.
0x04a: aa     MOV R2,A
0x04b: f9     MOV A,R1                            ; test 300+R4 bits
0x04c: f2 5c  JB7 loc_005c                        ; bit 7 do the set note data
0x04e: d2 73  JB6 loc_0073                        ; bit 6 set the delay & freq ?
0x050: b2 7d  JB5 loc_007d                        ; bit 5 silence
0x052: 92 84  JB4 loc_0084                        ; bit 4 tune finished.
0x054: b9 3f  MOV R1,##0x3f                       ; bits 4,5,6,7 all clear
0x056: f1     MOV A,@R1                           ; turn sound off again
0x057: 53 bf  ANL A,##0xbf                        ; clear bit 6 of Reg $3F
0x059: a1     MOV @R1,A
0x05a: 04 14  JMP loc_0014                        ; exit interrupt routine

loc_005c:
                                                ; *** BIT 7 SET ***
0x05c: 53 7f  ANL A,##0x7f                        ; bit 7 set
0x05e: ab     MOV R3,A                            ; R3 = Value in table
0x05f: b8 aa  MOV R0,##0xaa                       ; R0 = $AA
0x061: 27     CLR A
0x062: 90     MOVX @R0,A                          ; Clear ($AA) (sound off)
0x063: b8 a7  MOV R0,##0xa7                       ; R0 = $A7
0x065: b9 03  MOV R1,##0x03                       ; R1 = 3

loc_0067:
0x067: fa     MOV A,R2                            ; Read value from table
0x068: e3     MOVP3 A,@A                          ; ($300+2nd value)
0x069: 90     MOVX @R0,A                          ; Copying three bytes from
0x06a: 18     INC R0                              ; 2nd value in table to
0x06b: 1a     INC R2                              ; $A7,$A8,$A9
0x06c: e9 67  DJNZ R1,loc_0067
0x06e: fa     MOV A,R2                            ; 2nd Value = (300+2nd Value)
0x06f: e3     MOVP3 A,@A                          ; e.g. the next one
0x070: aa     MOV R2,A                            ; plonk it in R2
0x071: 04 76  JMP loc_0076

loc_0073:
                                                ; *** BIT 6 SET ***
0x073: 53 3f  ANL A,##0x3f                        ; mask out
0x075: ab     MOV R3,A                            ; plonk it in R3. (delay)

loc_0076:
0x076: 1c     INC R4                              ; skip the 2nd value

loc_0077:
0x077: b8 aa  MOV R0,##0xaa                       ; $(AA) is the last value
0x079: fa     MOV A,R2
0x07a: 90     MOVX @R0,A
0x07b: 04 14  JMP loc_0014                        ; continue the handler

loc_007d:
                                                ; *** BIT 5 SET ***
0x07d: 53 1f  ANL A,##0x1f                        ; set delay to bits 0..4
0x07f: ab     MOV R3,A
0x080: ba 00  MOV R2,##0x00                       ; volume = 0
0x082: 04 77  JMP loc_0077                        ; set it

loc_0084:
                                                ; *** BIT 4 SET ***
0x084: fa     MOV A,R2                            ; bit 4 is a 'jump'

loc_0085:
0x085: ac     MOV R4,A                            ; to the second value
0x086: 84 0a  JMP continuation_of_vblank          ; tune complete !
0x088: 00     NOP

ram_to_vdc_vblank_copying_check:
; ****************************************************************************
;
;       RAM to VDC Copier.
;       Called in VBlank Handler once if R3F Bit 7 is set. Bit is
;       cleared by the interrupt handler.
;
; ****************************************************************************
0x089: b8 7f  MOV R0,##0x7f                       ; R0 = $7F Copy Count
0x08b: b9 a0  MOV R1,##0xa0                       ; R1 = $A0 VDC Control
0x08d: 81     MOVX A,@R1                          ; Turn off Grid,Foreground
0x08e: 53 d6  ANL A,##0xd6
0x090: 91     MOVX @R1,A
0x091: 89 7c  ORL P1,##0x7c                       ; set : !kbscan !vdcen !ramen copyen
0x093: 99 e7  ANL P1,##0xe7                       ; clear : !vdcen !ramen

loc_0095:
0x095: 80     MOVX A,@R0                          ; if RAM($7F) is non-zero
0x096: 96 a3  JNZ copying_code                    ; jump to $A3
0x098: 89 bc  ORL P1,##0xbc                       ; set : !kbscan !vdcen !ramen  lumen
0x09a: 99 b7  ANL P1,##0xb7                       ; clear : !vdcen copyen
0x09c: b9 a0  MOV R1,##0xa0                       ; turn on grid & foreground
0x09e: 81     MOVX A,@R1
0x09f: 43 28  ORL A,##0x28
0x0a1: 91     MOVX @R1,A
0x0a2: 83     RET

copying_code:
; ****************************************************************************
;
;       R0 = ($7F), A = Byte Count. This is the active RAM->VDC Copying
;       Code
;
;               $7F = Count of Bytes
;               $7E = Address to Copy to.
;               $7D = Data (Backwards !)
;
; ****************************************************************************
0x0a3: aa     MOV R2,A                            ; save count in R2
0x0a4: c8     DEC R0                              ; point at address
0x0a5: 80     MOVX A,@R0                          ; R1 = Address.
0x0a6: a9     MOV R1,A                            ; put in r1
0x0a7: c8     DEC R0                              ; point to first data byte

loc_00a8:
0x0a8: 80     MOVX A,@R0                          ; copy from RAM to VDC
0x0a9: 91     MOVX @R1,A
0x0aa: c8     DEC R0                              ; previous copy from
0x0ab: 19     INC R1                              ; next copy to
0x0ac: ea a8  DJNZ R2,loc_00a8                    ; keep copyin' man
0x0ae: 04 95  JMP loc_0095                        ; exit.

keyboard_routine:
; ****************************************************************************
;
;                             Keyboard Check Routine
;
; ****************************************************************************
0x0b0: 15     DIS I
0x0b1: c5     SEL RB0
0x0b2: 09     IN A,P1                             ; read enable/switch status
0x0b3: ae     MOV R6,A                            ; save it in R6
0x0b4: 89 bc  ORL P1,##0xbc                       ; set : !kbscan !vdcen !ramen  lumen
0x0b6: 99 bb  ANL P1,##0xbb                       ; clear : !kbscan copyen
0x0b8: b8 f0  MOV R0,##0xf0                       ; R0 = Scan Row
0x0ba: ba 06  MOV R2,##0x06                       ; R2 = Rows to Scan

loc_00bc:
0x0bc: f8     MOV A,R0                            ; set the row counter
0x0bd: 3a     OUTL P2,A                           ; update keyboard scan row
0x0be: 0a     IN A,P2                             ; read keyboard status
0x0bf: 92 da  JB4 loc_00da                        ; no key pressed, try next
0x0c1: ba 30  MOV R2,##0x30                       ; r2 = 110000
0x0c3: a8     MOV R0,A                            ; r0 = scan value

loc_00c4:
0x0c4: 0a     IN A,P2                             ; read keyboard status
0x0c5: d8     XRL A,R0                            ; check debouncing
0x0c6: 96 df  JNZ loc_00df                        ; return none if a bounce
0x0c8: ea c4  DJNZ R2,loc_00c4
0x0ca: f8     MOV A,R0                            ; a = scan value
0x0cb: 47     SWAP A                              ; nibl exchange of A
                                                ; 4-6=row,0=press,1-3=col
0x0cc: 77     RRC A                               ; 0-2=col,3-5=row
0x0cd: d3 07  XRL A,##0x07                        ; and signed correctly ?
0x0cf: 53 3f  ANL A,##0x3f                        ; drop upper 2 bits
0x0d1: a9     MOV R1,A                            ; save in r1
0x0d2: ff     MOV A,R7                            ; r7 = Column & 3F
0x0d3: 53 3f  ANL A,##0x3f
0x0d5: d9     XRL A,R1                            ; check different to last one
0x0d6: 96 e5  JNZ loc_00e5                        ; if so, go to exit point
0x0d8: 04 df  JMP loc_00df                        ; same, return no key.

loc_00da:
0x0da: 18     INC R0                              ; no key pressed-next row
0x0db: ea bc  DJNZ R2,loc_00bc                    ; loop back
0x0dd: bf ff  MOV R7,##0xff                       ; r7 = a = FF

loc_00df:
0x0df: ff     MOV A,R7                            ; no key pressed
0x0e0: 43 c0  ORL A,##0xc0                        ; A = A or 1100 0000
0x0e2: af     MOV R7,A                            ; put in return reg... Failed
0x0e3: 64 78  JMP loc_0378                        ; key fail code

loc_00e5:
0x0e5: 64 76  JMP keyboard_in_routine_end         ; key succeed code

set_up_vdc_access:
; ****************************************************************************
;
;                             Set up the VDC access
;
; ****************************************************************************
0x0e7: 89 bc  ORL P1,##0xbc                       ; set : !kbscan !vdcen !ramen  lumen
0x0e9: 99 b7  ANL P1,##0xb7                       ; clear : !vdcen copyen (?)
0x0eb: 83     RET

set_up_ram_access:
; ****************************************************************************
;
;                             Set up the RAM access
;
; ****************************************************************************
0x0ec: 89 bc  ORL P1,##0xbc                       ; set : !kbscan !vdcen !ramen  lumen
0x0ee: 99 af  ANL P1,##0xaf                       ; clear : !ramen copyen
0x0f0: 83     RET

reset:
; ****************************************************************************
;
;                           Reset the Odyssey II
;
; ****************************************************************************
0x0f1: 89 bc  ORL P1,##0xbc                       ; set : !kbscan !vdcen !ramen  lumen
0x0f3: 99 b7  ANL P1,##0xb7                       ; clear : !vdcen copyen
0x0f5: 23 20  MOV A,##0x20                        ; clear registers $20 - $40
0x0f7: a8     MOV R0,A
0x0f8: a9     MOV R1,A
0x0f9: 27     CLR A

loc_00fa:
0x0fa: a0     MOV @R0,A
0x0fb: 18     INC R0
0x0fc: e9 fa  DJNZ R1,loc_00fa
0x0fe: 14 ec  CALL set_up_ram_access              ; access RAM
0x100: b9 ff  MOV R1,##0xff                       ; and clear that

loc_0102:
0x102: 91     MOVX @R1,A
0x103: e9 02  DJNZ R1,loc_0102
0x105: 91     MOVX @R1,A
0x106: 14 e7  CALL set_up_vdc_access              ; access the VDC Chip
0x108: 34 1c  CALL display_off                    ; turn off the display
0x10a: b8 ff  MOV R0,##0xff                       ; fill 80-FF with Zero
0x10c: b9 80  MOV R1,##0x80                       ; (clearing the grids)
0x10e: 27     CLR A

loc_010f:
0x10f: 90     MOVX @R0,A
0x110: c8     DEC R0
0x111: e9 0f  DJNZ R1,loc_010f
0x113: 23 f8  MOV A,##0xf8                        ; fill 00-7F with $F8

loc_0115:
0x115: 90     MOVX @R0,A                          ; move chars,sprites,quads
0x116: e8 15  DJNZ R0,loc_0115                    ; off screen
0x118: 90     MOVX @R0,A                          ; why not call the other one ?
0x119: 34 27  CALL display_on                     ; turn the display back on
0x11b: 83     RET

display_off:
; ****************************************************************************
;
;                         Turn off the Display
;
; ****************************************************************************
0x11c: 15     DIS I                               ; no interrupts
0x11d: c5     SEL RB0                             ; base register set
0x11e: b8 a0  MOV R0,##0xa0
0x120: 80     MOVX A,@R0                          ; read $A0 VDC Control Register

loc_0121:
0x121: 53 d6  ANL A,##0xd6                        ; Turn off foreground,grid
0x123: 90     MOVX @R0,A                          ; update VDC Control
0x124: d5     SEL RB1
0x125: 05     EN I                                ; interrupts back on
0x126: 83     RET

display_on:
; ****************************************************************************
;
;                           Turn the display back on
;
; ****************************************************************************
0x127: 15     DIS I                               ; int off
0x128: c5     SEL RB0                             ; base register set
0x129: b8 a0  MOV R0,##0xa0
0x12b: 80     MOVX A,@R0
0x12c: 43 28  ORL A,##0x28                        ; set Foreground,Grid on
0x12e: 90     MOVX @R0,A
0x12f: d5     SEL RB1
0x130: 05     EN I                                ; int back on
0x131: 83     RET

enable_data_copy_next_vsync:
; ****************************************************************************
;
;                     Enable a Data Copy at the next VSYNC
;
; ****************************************************************************
0x132: 27     CLR A                               ; Memory(R0) = 0
0x133: 90     MOVX @R0,A
0x134: b8 3f  MOV R0,##0x3f                       ; Set Bit 7 of r63.
0x136: f0     MOV A,@R0
0x137: 43 80  ORL A,##0x80
0x139: a0     MOV @R0,A

loc_013a:
0x13a: 14 e7  CALL set_up_vdc_access              ; access the VDC Chip
0x13c: 83     RET

get_keystroke:
; ****************************************************************************
;
;                               Get Keystroke into A
;
; ****************************************************************************
0x13d: 34 76  CALL wait_for_interrupt             ; Wait for Interrupt
0x13f: 14 b0  CALL keyboard_routine               ; Keyboard Check Routine
0x141: f2 3d  JB7 get_keystroke                   ; go back if no key pressed
0x143: 23 56  MOV A,##0x56                        ; R4 = 56,Set 3F bit 6
0x145: 34 a2  CALL start_tune                     ; play tune at $0356
0x147: c5     SEL RB0                             ; reselect page zero
0x148: ff     MOV A,R7                            ; get the result
0x149: d5     SEL RB1                             ; reselect page 1
0x14a: 83     RET

character_colour_translation:
; ****************************************************************************
;
;             R4 = Vertical R5 = Character. R6 = Colour Byte * 2.
;                 Convert for suitable use in VDC registers
;
; ****************************************************************************
0x14b: fe     MOV A,R6                            ; Bit 0 is part of the char set
0x14c: 53 fe  ANL A,##0xfe
0x14e: ae     MOV R6,A
0x14f: fd     MOV A,R5                            ; R5 = R5 * 8
0x150: 97     CLR C
0x151: f7     RLC A
0x152: f7     RLC A
0x153: f7     RLC A
0x154: ad     MOV R5,A
0x155: 27     CLR A                               ; A = 9th bit of character
0x156: f7     RLC A                               ; there are 64 characters
0x157: 4e     ORL A,R6                            ; or that bit into R6
0x158: ae     MOV R6,A
0x159: fc     MOV A,R4                            ; A = Vertical Position
0x15a: 67     RR A                                ; A = -Vertical / 2
0x15b: 37     CPL A                               ; presumably each pixel is two
0x15c: 17     INC A                               ; scanlines high.
0x15d: 6d     ADD A,R5                            ; Add it to the lower position
0x15e: ad     MOV R5,A                            ; subtract y/2 from char set pointer
0x15f: f6 6a  JC loc_016a                         ; if no borrow,exit
                                                ; else toggle bit 0 of R6
0x161: fe     MOV A,R6                            ; A = ~R6 & 1
0x162: 37     CPL A
0x163: 53 01  ANL A,##0x01
0x165: 2e     XCH A,R6                            ; swap A and R6
0x166: 53 fe  ANL A,##0xfe                        ; Mask off bits of R6
0x168: 4e     ORL A,R6                            ; or the borrow in
0x169: ae     MOV R6,A                            ; put it back - who wrote this ?

loc_016a:
0x16a: 83     RET

clear_all_characters:
; ****************************************************************************
;
;       Delete all characters by moving horizontal position to 248
;       which is off the right hand side.
;
; ****************************************************************************
0x16b: 23 f8  MOV A,##0xf8                        ; A = $F8
0x16d: b8 10  MOV R0,##0x10                       ; Start at VDC $10
0x16f: ba 30  MOV R2,##0x30                       ; Fill $30 characters

loc_0171:
0x171: 90     MOVX @R0,A
0x172: 18     INC R0
0x173: ea 71  DJNZ R2,loc_0171

loc_0175:
0x175: 83     RET

wait_for_interrupt:
; ****************************************************************************
;
;                             Wait for Interrupt
;
; ****************************************************************************
0x176: a5     CLR F1                              ; F1 = 0
0x177: 05     EN I                                ; Enable Interrupts

loc_0178:
0x178: 76 75  JF1 loc_0175                        ; Interrupt called ? Yes,exit
0x17a: 24 78  JMP loc_0178                        ; wait for interrupt

display_2_digit_bcd_characters:
; ****************************************************************************
;
;                Display BCD 2 digit A at VDC address R1.
;                R4 = Y R3 = X R6 = Colour
;
; ****************************************************************************
0x17c: 14 ec  CALL set_up_ram_access              ; access RAM only
0x17e: aa     MOV R2,A                            ; save A in R2
0x17f: b8 7f  MOV R0,##0x7f
0x181: 23 08  MOV A,##0x08                        ; Copy 8 bytes ($7F) = 9
0x183: 90     MOVX @R0,A
0x184: c8     DEC R0

loc_0185:
0x185: f9     MOV A,R1                            ; Address in R1
0x186: 90     MOVX @R0,A                          ; To VDC Address R1 ($7E)
0x187: c8     DEC R0                              ; R0 = $7D First Data Byte
0x188: fa     MOV A,R2                            ; R2 = First Data Byte (low)
0x189: 47     SWAP A                              ; A = First Data Byte (High)
0x18a: 53 0f  ANL A,##0x0f                        ; R5 = High Nibble
0x18c: ad     MOV R5,A                            ; put in R5
0x18d: 34 97  CALL loc_0197                       ; write it
0x18f: fa     MOV A,R2                            ; Write Low Nibble
0x190: 53 0f  ANL A,##0x0f
